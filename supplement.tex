\begin{abstract}
This supplement carries what the paper's thirteen pages could not hold: the
boundary-complete proofs behind the sketches in Section III, the accounting
details of every experiment in Section V, and five tables reporting in full
what the paper states only in aggregate---the adversarial injection
taxonomy, the complete detection--cost curve, the fold-by-fold selection
grid, ordering outcomes by intent class, and the frozen result table.
Nothing here revises a claim in the paper; each section supplies the detail
behind one.  Section~\ref{sec:restate} restates the definitions and results
the proofs refer to, so this document can be read without the paper open;
equation and theorem numbers here are LOCAL, and the correspondence to the
paper's numbering is given in Table~\ref{tab:numbering}.
\end{abstract}

\section{Restated Definitions and Results}
\label{sec:restate}

Throughout, one verification profile \((e,m)\) and one deterministic constant
budget \(b\) are fixed.  Write \(q\) for the probability that an
intent-fulfilling candidate plan violates the common loss event
\(\neg\Safe_{\prefix}\), \(p(b)\) for the probability that the verifier
rejects such a plan, \(\tau(b)\) for the deterministic verification-time
reservation, \(G\) for the time to obtain one candidate, and \(C_v,C_t>0\)
for the violation and time costs.  Let \(V(b)\) be the probability that the
eventually actuated plan is prefix-unsafe and \(T(b)\) the
reservation-accounted completion time.

The inner optimization is
\begin{equation}
 \min_{b\in[0,b_{\max}]}
 J(b):=C_v V(b)+C_t\E[T(b)] .
\label{eq:objective}
\end{equation}

\begin{proposition}[Risk--latency frontier]
\label{prop:renewal}
Let \(r(b)=qp(b)<1\) and let \(N\) be the number of
generation--verification rounds before exit.  Then
\begin{equation}
 \E[N]=\frac{1}{1-r(b)},\quad
 V(b)=\frac{q[1-p(b)]}{1-qp(b)},
\label{eq:renewal}
\end{equation}
\begin{equation}
 \E[T(b)]=\frac{G+\tau(b)}{1-qp(b)},\quad
 S_{\rm eff}(b):=\E[T(b)].
\label{eq:etime}
\end{equation}
\(V\) is strictly decreasing and \(\E[T]\) strictly increasing wherever
\(p'(b)>0\).
\end{proposition}

The closed-form results assume the exponential--linear family
\begin{equation}
 p(b)=\bar p(1-e^{-\alpha b}),\qquad
 \tau(b)=\tau_0+\beta b,
\label{eq:exp-linear}
\end{equation}
with \(\alpha,\beta>0\) and a strict detection ceiling \(\bar p\in(0,1)\).
Define
\begin{equation}
 H:=G+\tau_0,\qquad
 \delta:=\frac{\beta}{\bar p\alpha},\qquad
 R:=\frac{C_v}{C_t}.
\label{eq:constants}
\end{equation}
Section V-C of the paper reports that the detection half of
\eqref{eq:exp-linear} does \emph{not} fit the measured curves; it is the
analytical example that yields the closed forms below, not a fitted law.

Multiplying \(J'\) by its positive denominator gives the sign-carrying
quantity used throughout:
\begin{align}
 A(b)&:=J'(b)[1-qp(b)]^2=K+e^{-\alpha b}g(b),\label{eq:Ab}\\
 K&:=\beta C_t(1-q\bar p)>0,\nonumber\\
 g(b)&:=q\bar p\{\alpha[(q-1)C_v+(H+\beta b)C_t]+\beta C_t\}.\nonumber
\end{align}

\begin{theorem}[Single crossing and participation]
\label{thm:single-crossing}
On \(b\geq0\), \(J\) is either strictly increasing for \(b>0\) with
minimizer \(b_\infty^\star=0\), or it strictly decreases to one interior
minimizer and then strictly increases.  An interior budget is selected iff
\begin{equation}
 b_\infty^\star>0
 \iff
 R>F(q):=\frac{H+\delta/q}{1-q}.
\label{eq:participation}
\end{equation}
At \(R=F(q)\), \(J'(0)=0\) and \(J'(b)>0\) for every \(b>0\).
\end{theorem}

\begin{corollary}[Lambert-\(W\) solution and clipping]
\label{cor:lambert}
With
\(c_0:=\alpha[(q-1)C_v+HC_t]+\beta C_t\),
\(c_1:=\alpha\beta C_t\), and
\(\theta:=\frac{\alpha K}{q\bar p c_1}\exp(-\alpha c_0/c_1)>0\),
the unbounded optimum in the active region is
\begin{equation}
 b_\infty^\star=-\frac{c_0}{c_1}-\frac{1}{\alpha}W_0(\theta),
\label{eq:lambert}
\end{equation}
and the deployed policy is
\begin{equation}
 b_D^\star=\min\{b_\infty^\star,b_{\max}\}.
\label{eq:clipping}
\end{equation}
\end{corollary}

\begin{theorem}[Worthwhile interval and error-rate shape]
\label{thm:q-shape}
\(F\) is strictly U-shaped on \((0,1)\), with
\(q_0=\sqrt{\delta}/(\sqrt{H+\delta}+\sqrt{\delta})\) and
\(F_{\min}=(\sqrt{H+\delta}+\sqrt{\delta})^2\).  If \(R\leq F_{\min}\) then
\(b_\infty^\star(q)=0\) for every \(q\).  If \(R>F_{\min}\), define
\begin{equation}
 q_\pm=\frac{(R-H)\pm\sqrt{(R-H)^2-4R\delta}}{2R}.
\label{eq:qroots}
\end{equation}
Then \(0<q_-<q_+<1\), \(b_\infty^\star(q)>0\) iff \(q\in(q_-,q_+)\), and
\(b_\infty^\star\) is strictly increasing then strictly decreasing there.
\end{theorem}

For the queueing result, \(\mathcal M(Q):=\arg\min_{b\in[0,b_{\max}]}
\{C_vV(b)+QS_{\rm eff}(b)\}\) is the backlog-weighted static
correspondence, nonincreasing in the strong set order.
